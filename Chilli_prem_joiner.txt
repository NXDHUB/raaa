local function cleanMoney(val)
    if type(val) == "number" then return val end
    if type(val) == "string" then
        local cleaned = val:gsub("[^0-9%.]", "")
        return tonumber(cleaned) or 0
    end
    return 0
end

local function getNormalizedMoney(val)
    val = cleanMoney(val)
    if val > 999999 then
        return val / 1e6
    end
    return val
end

local function formatMoney(val)
    val = getNormalizedMoney(val)
    if val >= 1000 then
        return string.format("%.1fB", val / 1000)
    end
    return string.format("%.0fM", val)
end

    local function getGuiRoots()
        local roots = {CoreGui, LocalPlayer:WaitForChild("PlayerGui")}
        pcall(function() if gethui then table.insert(roots, gethui()) end end)
        return roots
    end

    local function searchForElements(root)
        for _, v in ipairs(root:GetDescendants()) do
            if (v:IsA("TextLabel") or v:IsA("TextButton")) and v.Text then
                local txt = string.lower(v.Text)
                if txt:match("join job%-id") then
                    if v:IsA("TextLabel") and v.Parent then
                        targetBtn = v.Parent
                    else
                        targetBtn = v
                    end
                end
            end
            if v:IsA("TextLabel") and v.Text then
                local txt = string.lower(v.Text)
                if txt:match("job%-id input") then
                    local parent = v.Parent
                    if parent then
                        for _, sib in ipairs(parent:GetDescendants()) do
                            if sib:IsA("TextBox") then
                                targetBox = sib
                            end
                        end
                        if not targetBox and parent.Parent then
                            for _, sib in ipairs(parent.Parent:GetDescendants()) do
                                if sib:IsA("TextBox") then
                                    targetBox = sib
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    for _, root in ipairs(getGuiRoots()) do
        pcall(function() searchForElements(root) end)
        if targetBox and targetBtn then break end
    end

    if targetBox and targetBtn then
        pcall(function()
            targetBox.Text = tostring(id)
            local ok, conns = pcall(getconnections, targetBox.FocusLost)
            if ok and conns then
                for _, c in ipairs(conns) do
                    if c.Function then task.spawn(c.Function, true) end
                end
            end
        end)

        task.wait()

        pcall(function()
            if targetBtn:IsA("GuiButton") then
                local events = {targetBtn.MouseButton1Click, targetBtn.MouseButton1Down, targetBtn.Activated}
                for _, event in ipairs(events) do
                    local ok, conns = pcall(getconnections, event)
                    if ok and conns then
                        for _, c in ipairs(conns) do
                            if c.Function then task.spawn(c.Function) end
                        end
                    end
                end
            else

                        if c.Function then
                            task.spawn(c.Function, {UserInputType = Enum.UserInputType.MouseButton1})
                        end
                    end
                end
            end
        end)
        return true
    end

    return false

    local moneyValM = getNormalizedMoney(rawMoney)

        return

    if isTargetMode then
        if not state.targetList[name] then return end
        local lim = state.targetLimits[name] or 0

local function ConnectWS()
    local HttpService = game:GetService("HttpService")
    local success, connect = pcall(function()
        return (syn and syn.websocket and syn.websocket.connect)
            or (WebSocket and WebSocket.connect)
            or (websocket and websocket.connect)
    end)
    if not success or not connect then
        StatusText.Text = "Incompatible Executor"
        return
    end

    local WS_URL = "wss://saintfinder-port-websocket-production.up.railway.app/ws"

    local function loop()
        local success, ws = pcall(function()
            return connect(WS_URL)
        end)
        if success and ws then
            StatusText.Text = "Connected To Secure Servers"
            ws.OnMessage:Connect(function(msg)
                local ok, decoded = pcall(HttpService.JSONDecode, HttpService, msg)
                if ok and decoded and decoded.money then
                    AddEntry(decoded.name, decoded.money, decoded.jobid, decoded.players)
                end
            end)
        else
            StatusText.Text = "Retrying"
            task.wait(3)
            loop()
        end
    end

    loop()
end

ConnectWS()
